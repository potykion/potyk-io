function tickerKey(t) {
  if (!t) return "—";
  if (typeof t === "object") {
    // Link без |alias: display пустой, а fileName — метод (не строка).
    // Раньше return t.fileName отдавал саму функцию → в h3 был её исходник.
    if (t.display) return String(t.display);
    if (t.path) return String(t.path).split("/").pop().replace(/\.md$/i, "");
    if (typeof t.fileName === "function") {
      try { return String(t.fileName()); } catch (_) {}
    }
    if (typeof t.fileName === "string" && t.fileName) return t.fileName;
    return String(t);
  }
  const s = String(t);
  const pipe = s.match(/\|([^\]]+)\]\]/);
  if (pipe) return pipe[1];
  const path = s.match(/\[\[([^\]]+)\]\]/);
  if (path) return path[1].split("/").pop();
  return s;
}

function fmtDate(d) {
  if (!d) return "";
  const dt = dv.date(d);
  return dt ? dt.toFormat("dd.MM.yyyy HH:mm") : String(d);
}

function fmtVal(v) {
  if (v == null || v === "") return null;
  return String(v).trim() || null;
}

function sentimentBg(s) {
  const v = fmtVal(s) || "";
  if (v.includes("🟢")) return { bg: "rgba(34,197,94,0.18)", border: "rgba(34,197,94,0.45)" };
  if (v.includes("🟡")) return { bg: "rgba(234,179,8,0.20)", border: "rgba(234,179,8,0.50)" };
  if (v.includes("🔴")) return { bg: "rgba(239,68,68,0.18)", border: "rgba(239,68,68,0.45)" };
  return { bg: "var(--background-secondary)", border: "var(--background-modifier-border)" };
}

function tickerStem(key) {
  return String(key || "").split("/").pop().split("|")[0].trim();
}

function resolveTickerPage(t) {
  if (t && typeof t === "object" && t.path) {
    const byPath = dv.page(t.path);
    if (byPath) return byPath;
  }
  const key = tickerStem(tickerKey(t));
  if (!key) return null;
  const code = key.split(/\s+/)[0];
  return dv.pages('"Тикеры"')
    .where(p => {
      const name = p.file.name;
      return name === key || name === code || name.startsWith(code + " ");
    })
    .first();
}

function propLabel(raw) {
  if (raw == null || raw === "") return "";
  const list = Array.isArray(raw) ? raw : [raw];
  return list.map(x => String(x).trim()).filter(Boolean).join(", ");
}

function sectorOf(tickerPage) {
  if (!tickerPage) return "";
  return propLabel(tickerPage["Сектор"]);
}

function depsOf(tickerPage) {
  if (!tickerPage) return "";
  return propLabel(tickerPage["Зависимости"]);
}

const FOLDER = "Новости";
const inFolder = (path) => path === FOLDER || path.startsWith(FOLDER + "/");

const pages = dv.pages()
  .where(p => p.ticker && inFolder(p.file.folder))
  .sort(p => p.datetime ?? p.file.ctime, "desc");

const groups = Array.from(pages.groupBy(p => tickerKey(p.ticker)))
  .map(g => {
    const page = resolveTickerPage(g.rows[0].ticker);
    return {
      key: g.key,
      rows: g.rows,
      sector: sectorOf(page),
      deps: depsOf(page),
    };
  })
  .sort((a, b) => {
    const aEmpty = !a.sector;
    const bEmpty = !b.sector;
    if (aEmpty !== bEmpty) return aEmpty ? -1 : 1;
    const bySector = a.sector.localeCompare(b.sector, "ru");
    if (bySector) return bySector;
    return String(a.key).localeCompare(String(b.key), "ru");
  });

const root = dv.container.createDiv({ cls: "news-cards-root" });
root.style.cssText = "width:100%;max-width:100%;display:block;clear:both;";

let lastSector = null;
for (const group of groups) {
  const sectorTitle = group.sector || "Без сектора";
  if (sectorTitle !== lastSector) {
    const sectorEl = root.createEl("h2", { text: sectorTitle });
    sectorEl.style.cssText = "margin:1.5rem 0 0.5rem;opacity:0.85;";
    lastSector = sectorTitle;
  }

  const section = root.createDiv({ cls: "news-cards-section" });
  section.style.cssText = "width:100%;display:block;margin:0 0 1.25rem;";

  const heading = section.createEl("h3");
  heading.append(String(group.key));
  if (group.deps) {
    const hint = heading.createSpan({ text: ` · ${group.deps}` });
    hint.style.cssText = "font-weight:400;opacity:0.65;font-size:0.85em;";
  }

  const grid = section.createDiv({ cls: "news-cards-grid" });
  grid.style.cssText = [
    "display:flex",
    "flex-wrap:nowrap",
    "gap:1rem",
    "width:100%",
    "margin:0.75rem 0 0",
    "overflow-x:auto",
    "overflow-y:hidden",
    "padding-bottom:0.35rem",
    "-webkit-overflow-scrolling:touch",
  ].join(";");

  for (const p of group.rows) {
    const tone = sentimentBg(p.sentiment);
    const card = grid.createDiv({ cls: "news-card" });
    card.style.cssText = [
      `border:1px solid ${tone.border}`,
      "border-radius:8px",
      "padding:0.9rem 1rem",
      `background:${tone.bg}`,
      "flex:0 0 280px",
      "width:280px",
      "max-width:280px",
      "min-width:280px",
      "box-sizing:border-box",
    ].join(";");

    const title = card.createDiv();
    title.style.cssText = "font-weight:600;margin-bottom:0.35rem;";
    dv.el("span", p.file.link, { container: title });

    if (p.datetime) {
      const dateEl = card.createDiv();
      dateEl.style.cssText = "font-size:0.85em;opacity:0.7;margin-bottom:0.45rem;";
      dateEl.setText(fmtDate(p.datetime));

    const meta = [fmtVal(p.sentiment), fmtVal(p.price)].filter(Boolean);
    if (meta.length) {
      const metaEl = card.createDiv();
      metaEl.style.cssText = "font-size:0.9em;margin-bottom:0.6rem;opacity:0.9;";
      metaEl.setText(
        [fmtVal(p.sentiment) && `sentiment: ${fmtVal(p.sentiment)}`, fmtVal(p.price) && `price: ${fmtVal(p.price)}`]
          .filter(Boolean)
          .join(" · ")
      );
    }

    const bodyEl = card.createDiv();
    bodyEl.style.cssText = "font-size:0.92em;line-height:1.45;white-space:pre-wrap;overflow-wrap:anywhere;";
    bodyEl.setText(fmtVal(p.summary) || "—");
  }
}
